      private assertTransition(
        from: SubscriptionStatus,
        to: SubscriptionStatus,
        by: TransitionActor,
        reason: string | undefined,
        planType?: string,

          analysis: JSON.stringify({ from: sub.status, to: status, by: actorId, actorRole, reason: reason ?? null }),
          recommendation: "",
        } });
        return updated;
      });
