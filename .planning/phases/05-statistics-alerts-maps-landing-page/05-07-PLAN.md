    ```
    "use client";
    import { useEffect, useState } from "react";
    import { io, Socket } from "socket.io-client";

    type Scope = 'national' | 'viloyat' | 'tuman' | 'mahalla';
    export function useStatsSocket(scope: Scope, scopeId: string) {
      const [payload, setPayload] = useState<any>(null);
      const [connected, setConnected] = useState(false);
      const [error, setError] = useState<Error | null>(null);

      useEffect(() => {
        const url = process.env.NEXT_PUBLIC_API_URL;
        if (!url) return; // dev no-op when unset
        const socket: Socket = io(url, { withCredentials: true });
        const room = `stats:${scope}:${scopeId}`;
        socket.on('connect', () => { setConnected(true); socket.emit('subscribe', { room }); });
        socket.on('stats:refresh', (msg: any) => setPayload(msg.payload));
        socket.on('connect_error', (err) => setError(err));
        socket.on('disconnect', () => setConnected(false));
        return () => {
          socket.emit('unsubscribe', { room });
          socket.disconnect();
        };
      }, [scope, scopeId]);

      return { payload, connected, error };
    }
    ```
