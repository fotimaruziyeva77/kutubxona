    ```
    "use client";
    import { useQuery } from "@tanstack/react-query";

    type LandingStats = { libraries: number; users: number; books: number; transactionsThisMonth: number; readingIndex: number; lastUpdated: string | null };

    export function LiveStatsRefresher({
      initialStats,
      slot,
    }: {
      initialStats: LandingStats;
      slot: 'libraries' | 'users' | 'books' | 'tx' | 'ri';
    }) {
      const { data } = useQuery({
        queryKey: ['landing-live'],
        queryFn: async () => {
          const r = await fetch(`${process.env.NEXT_PUBLIC_API_URL}/landing/stats`);
          if (!r.ok) throw new Error('fetch failed');
          return r.json() as Promise<LandingStats>;
        },
        initialData: initialStats,
        refetchInterval: 30_000,
        refetchIntervalInBackground: false,
      });
      const value = slot === 'libraries' ? data.libraries :
                    slot === 'users' ? data.users :
                    slot === 'books' ? data.books :
                    slot === 'tx' ? data.transactionsThisMonth :
                    data.readingIndex;
      return <span>{new Intl.NumberFormat("uz-UZ").format(Math.round(value))}+</span>;
    }
    ```
